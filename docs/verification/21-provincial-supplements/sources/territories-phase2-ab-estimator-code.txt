// formats a number as currency
// @param num   the number to format
// returns      num formatted as currency with sign and decimals
function formatCurrency(num) {
    num = num.toString().replace(/\$|\,/g, '');
    if (isNaN(num)) num = "0";
    sign = (num == (num = Math.abs(num)));
    num = Math.floor(num * 100 + 0.50000000001);
    cents = num % 100;
    num = Math.floor(num / 100).toString();
    if (cents < 10) cents = "0" + cents;
    for (var i = 0; i < Math.floor((num.length - (1 + i)) / 3); i++)
    num = num.substring(0, num.length - (4 * i + 3)) + ',' + num.substring(num.length - (4 * i + 3));
    return (((sign) ? '' : '-') + '$' + num + '.' + cents);
}

    }
    $('#applicantIncomeForCalc').val(myTotal);
}

// calculates Supplementary Accommodation Benefit (SAB), if applicable
// two types of SAB: SAB1, SAB2. SAB1 exists only in grandfathering situations post-2013. Estimator only deals with SAB2 as it
//   cannot determine whether an applicant is/should be grandfathered without personal information
// was named getSABType; renamed for clarity
function estimateSAB(isSingle) {
    if (isSingle) {
        var sabIncome = parseInt($('.applicantTotalIncome').val());
    } else {
        var sabIncome = (parseInt($('.spouseTotalIncome').val()) + parseInt($('.applicantTotalIncome').val())) / 2;
    }
    var thisSAB = 0;
    thisSAB = (monthlyPrivateRoomRate + monthlyDiscretionaryIncome) - (sabIncome / 12);
    //Debug
    if (isDebug)  alert("Room Rate: (" + monthlyPrivateRoomRate + " + Dis Rate:" + monthlyDiscretionaryIncome + ")" + "- (Income: " + sabIncome + "/ 12) = thisSAB: " + thisSAB);
    if (thisSAB > sabType2MonthlyMax) {
        thisSAB = sabType2MonthlyMax;
    }
    if (thisSAB < 0) {
        thisSAB = 0;
    }
    if (thisSAB > 0 && thisSAB < 10) {
        thisSAB = 10;
    }

    return Math.floor(thisSAB);
}

// Performs the Dental Eligibility Calculation
//  Eligibility % = (DASP Max - (Total Adjusted Income - Previous SAB)) / 35.25
function calcDental(householdIncome) {

    dental = 0;
    var lIncomeThresholdDental = 0;
    var uIncomeThresholdDental = 0;
    var thresholdOverIncome = false;
    //MARRIED VS SINGLE --------------------------

   //if (maritalStatus == 0) {
    if ($('.maritalStatus').val() == 0) {
        lIncomeThresholdDental = dentalSingleIncomeLowerRangeMax;
        uIncomeThresholdDental = dentalSingleIncomeMidRangeMax;
    } else {
        lIncomeThresholdDental = dentalCoupleIncomeLowerRangeMax;
        uIncomeThresholdDental = dentalCoupleIncomeMidRangeMax;
    }
    //DENTAL CHECKS ------------------------------
    //Upper income - total income / upper income - lower income * 100
    if (householdIncome > uIncomeThresholdDental) {
        dental = 0;
    } else {
        dental = ((uIncomeThresholdDental - householdIncome) / (uIncomeThresholdDental - lIncomeThresholdDental) * 100);
    }
  //if (isDebug)  alert("("+uIncomeThresholdDental +"-"+ householdIncome+") / ("+uIncomeThresholdDental+" -"+ lIncomeThresholdDental+") * 100) = "+ dental + " maritalStatus " + $('.maritalStatus').val());
    //Range Checks
    if (dental > 100) {
        dental = 100;
    }
    if (dental <= 9.99 && dental > 0) {
        dental = 10;
    }
    if (dental < 0) {
        dental = 0;
    }
    return Math.floor(dental);
}

// was labeled calcOASFormula()
// renamed because it's actually calculating Optical Assistance amount, not Old Age Security!
function calcOptical(dentalAmount) {
    if ((dentalAmount >= 10) && (dentalAmount < 100)) {
        opticalAssistance = 115;
    } else if (dentalAmount == 100) {
        opticalAssistance = 230;
    } else {
        opticalAssistance = 0;
    }
    return opticalAssistance;
}

// Performs the Basic Monthly Cash Benefit Calculation outlined at the end of the ASB Maximum Benefits & Phase-out Rates table
//   which is described as (Applicable Maximum Cash Benefit - (NDI*phase-out))/ 12
// was called calcFormula; renamed for clarity
function calcASB(thisMaxBenefit, NDI, phaseOutRate, reductionRate) {
    //Debug
    if (isDebug)  alert('Max Ben:' + thisMaxBenefit + " NDI:" + NDI + " phase rate:" + phaseOutRate + " reduction:" + reductionRate);
    var stepOne = NDI;
    if (NDI > 0) {
        stepOne = NDI * phaseOutRate;
    }

    var stepTwo = thisMaxBenefit - stepOne;
    if (reductionRate != 0) {
        if (isDebug)  alert("Caught Reduction rate");
        //Dont multiply by 0
        stepTwo = stepTwo * reductionRate;
    }
    totalApp = stepTwo / 12;

    if (isDebug)  alert("Calculation: [" + thisMaxBenefit + "- (" + NDI + " * " + phaseOutRate + ")] / 12 = " + totalApp);
    return totalApp;
}

                                         if (isDebug) { console.log('reCaptcha v3 score too low, showing v2 challenge') };

//console.log("DEBUG token " + token  + " HN1 " + captchaHostname.indexOf("seniors-housing.alberta.ca") + " HN2 " + captchaHostname.indexOf("seniors-housing.alberta.ca") + " checkbox-expired " + isCaptchaChecked());

             // if captcha already done and valid, skip v2
                                                if (captchaSuccess && token && (token != "") && ((captchaHostname.indexOf("seniors-housing.alberta.ca") == 0) || (captchaHostname.indexOf("seniors-housing.alberta.ca") == 4)) && (captchaAction == "captcha") && isCaptchaChecked() ) {
                                                        //console.log("captcha- this is a back and forth, we have a valid NaR, make the call");
                                                         // captcha success, proceed as normal
                                                        estimateBenefitAmounts(maritalStatus, applicantRescode, applicantIncome, appOAS, spouseRescode, spouseIncome, sppOAS);
                                                        callEstimateWS();
                                                } else {
              // hide results form until captcha dealt with
                                                        $('#wizard-form-p-4').hide();
                                                        $('#recaptchaError').html('<b>Your submission has been selected for additional verification:</b><ul><li>Please check the "I\'m Not a Robot" box below.</li></ul>');
                                                        $('#recaptchaError').show();
                                                        $('.g-recaptcha').show();

                                                        // scroll to error messages
              $(document).scrollTop($('#main').offset().top - $('#main').height());
                                                }

                                        //alert("CAPTCHA failed. Please reload the estimator and try again. If problems persist, please try again later.");
                                        //$('#wizard-form-t-3').click();
                                } else {
                                        // captcha success, proceed as normal
                                        estimateBenefitAmounts(maritalStatus, applicantRescode, applicantIncome, appOAS, spouseRescode, spouseIncome, sppOAS);
                                        callEstimateWS();
                                }
                                //console.log("captcha success ' + captchaSuccess + " captcha score " + captchaScore);
                        });

                });
}

// ARMS 43293 For the combination of DSL and LTC into the single option of "Continuing Care"
// returns true if the individual is in continuing care, false otherwise
// send in either "applicant" or "spouse" as the individual to check
// LTC/DSL codes are "N" and "A", and maybe "C" for Contuining Care
function isInContinuingCare(individual) {

  var selectedResidence = $('.'+individual+'Residence option:selected').val();
  console.log("IsInContinuingCare individual " + individual + " val " + selectedResidence);

  if (selectedResidence == "N" || selectedResidence == "A" || selectedResidence == "C") {
    return true;
  } else {
    return false;
  }

}
